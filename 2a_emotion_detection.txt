import requests

def emotion_detector(text_to_analyze):
    """
    Runs emotion detection on the provided text using the Watson NLP library & returns the raw text response.
    """
    # Define the URL - Emotion Detection service
    url = 'https://sn-watson-emotion.labs.skills.network/v1/watson.runtime.nlp.v1/NlpService/EmotionPredict'
    
    # Set up the headers as required by API
    headers = {"grpc-metadata-mm-model-id": "emotion_aggregated-workflow_lang_en_stock"}
    
    # Create the payload with analyze
    myobj = { "raw_document": { "text": text_to_analyze } }
    
    # Send the POST request to emotion detection service
    response = requests.post(url, json=myobj, headers=headers)
    
    # Return the attribute of the response object
    return response.text
